=== chippenham.pyfun ===
# Direct trains from a station, grouped by destination — a Network Rail SCHEDULE
# feed turned into a readable Markdown timetable, with NO Python helper: the file
# is streamed, parsed, and rendered entirely in Pyfun.
#
#   cargo run -- run examples/interop/network-rail/chippenham.pyfun
#
# This is the boundary-vs-engine interop pattern taken to its limit — even the
# streaming boundary is Pyfun. The file object from `open`/`gzip.open` is consumed
# as a lazy `Seq string`, so a ~3 GB NDJSON snapshot streams in constant memory; a
# substring prefilter skips JSON parsing on the lines that can't match; and the
# `Decode` module turns each surviving line into typed data — no dict-poking, no
# bespoke Python. `open`/`gzip.open`'s mode + encoding and `write_text`'s encoding
# + newline are pinned as `extern` keyword arguments (see the externs below), so the Python
# call carries `encoding="utf-8"` without any positional filler.
#
# Runs out of the box against the bundled `sample.ndjson`. For the real feed
# (register at https://opendata.nationalrail.co.uk / Network Rail data feeds),
# point `dataFile` at your snapshot — the EXACT filename, `.json` or `.json.gz`
# (a `.gz` is detected by extension and streamed without decompressing).
#
# Performance: driving the ~20M-line loop through `Seq.fold` is correctness- and
# clarity-first, not throughput-first — expect a minute or two on the full feed
# (a native Python loop is faster, but that puts the hot path back in Python).
#
# STP caveat: counts reflect the day-of-week pattern of permanent + new schedules
# minus full cancellations (stp "C"), deduped by (departure, destination). A
# specific calendar date may differ. A full snapshot is all `Create` records, so
# transaction type isn't checked.

# `mode="rt"` opens the file as text and `encoding="utf-8"` reads UTF-8 instead of
# the platform locale (cp1252 on Windows) — both correct for the feed and ~5s
# faster on the full file. Pinning them as `extern` keyword arguments reaches
# `encoding` (a keyword-only / deep-positional arg) with no positional filler.
extern openText : string -> Seq string = builtins.open(mode="rt", encoding="utf-8")
extern gzipText : string -> Seq string = gzip.open(mode="rt", encoding="utf-8")
extern type Path
extern toPath : string -> Path = pathlib.Path
extern writeText : Path -> string -> int = pathlib.Path.write_text(encoding="utf-8", newline="\n")

type Stop = { tiploc: string, arr: string, dep: string }
type Run  = { uid: string, days: string, stp: string, stops: List Stop }
type Leg  = { dest: string, finalDest: string, rdays: string, depMin: int, arrMin: int, jt: int, via: List string }

let dataFile = "examples/interop/network-rail/sample.ndjson"
let outFile  = "examples/interop/network-rail/chippenham-routes.md"

# --- streaming boundary, in Pyfun: the file object is a lazy Seq of lines ---
let lines path =
  if String.endsWith ".gz" path then gzipText path else openText path

# --- decode the relevant lines into typed data (parse, don't validate) ---
let optStr key = Decode.oneOf [Decode.field key Decode.string, Decode.succeed ""]

let tipDec =
  Decode.field "TiplocV1"
    (Decode.map2 (fun c n -> (c, n))
      (Decode.field "tiploc_code" Decode.string)
      (optStr "tps_description"))

let stopDec =
  Decode.map3 (fun t a d -> (t, a, d))
    (Decode.field "tiploc_code" Decode.string)
    (optStr "public_arrival")
    (optStr "public_departure")

let schedDec =
  Decode.field "JsonScheduleV1"
    (Decode.map4 (fun uid days stp stops -> (uid, days, stp, stops))
      (Decode.field "CIF_train_uid" Decode.string)
      (optStr "schedule_days_runs")
      (optStr "CIF_stp_indicator")
      (Decode.field "schedule_segment" (Decode.field "schedule_location" (Decode.list stopDec))))

# Fold the lazy line Seq into (tiploc names, Chippenham-calling runs). A cheap
# substring test skips JSON parsing on the millions of lines that can't match —
# the same prefilter the Python did, now in Pyfun and in constant memory.
let ingest acc line =
  match acc:
    case (names, runs):
      if String.contains "TiplocV1" line then
        match Decode.decodeString tipDec line:
          case Ok (c, n): (Map.add c n names, runs)
          case Error _: acc
      else if String.contains "JsonScheduleV1" line and String.contains "CHIPNHM" line then
        match Decode.decodeString schedDec line:
          case Ok r: (names, List.concat runs [r])
          case Error _: acc
      else acc

let scan path = Seq.fold ingest (Map.empty, []) (lines path)

# --- shape the decoded tuples into typed records ---
let toStop t =
  match t:
    case (tp, a, d): Stop { tiploc = tp, arr = a, dep = d }

let toRun r =
  match r:
    case (uid, days, stp, stops):
      Run { uid = uid, days = days, stp = stp, stops = List.map toStop stops }

# A stop is a public calling point iff it advertises a public time.
let calls s = (s.arr != "" and s.arr != "0000") or (s.dep != "" and s.dep != "0000")

let toMin hhmm =
  match (String.toInt (String.slice 0 2 hhmm), String.toInt (String.slice 2 4 hhmm)):
    case (Some h, Some m): h * 60 + m
    case _: -1

let fmt2 n = if n < 10 then f"0{n}" else f"{n}"
let showTime m = f"{fmt2 (m // 60)}:{fmt2 (m % 60)}"

let jtime dep arr =
  let d = arr - dep
  if d < 0 then d + 1440 else d

let fromChippenham stops =
  match stops:
    case []: None
    case [s, *rest]:
      if s.tiploc == "CHIPNHM" and calls s then Some (s.dep, rest)
      else fromChippenham rest

let lastTiploc stops =
  match stops:
    case []: ""
    case [s]: s.tiploc
    case [_, *rest]: lastTiploc rest

let walkLegs days term depM via stops =
  match stops:
    case []: []
    case [s, *rest]:
      if calls s then
        let arrM = toMin s.arr
        let onward = List.concat via [s.tiploc]
        if arrM < 0 then walkLegs days term depM onward rest
        else
          let leg = Leg { dest = s.tiploc, finalDest = term, rdays = days, depMin = depM,
                          arrMin = arrM, jt = jtime depM arrM, via = via }
          List.concat [leg] (walkLegs days term depM onward rest)
      else walkLegs days term depM via rest

let legsOfRun r =
  if r.stp == "C" then []
  else
    match fromChippenham r.stops:
      case None: []
      case Some (chippDep, downstream):
        let depM = toMin chippDep
        if depM < 0 then []
        else walkLegs r.days (lastTiploc downstream) depM [] downstream

let runsOn day days = String.slice day (day + 1) days == "1"

let dedupLegs legs =
  let step acc leg =
    match acc:
      case (seen, out):
        let k = f"{leg.depMin}|{leg.dest}"
        match Map.tryFind k seen:
          case Some _: (seen, out)
          case None: (Map.add k true seen, List.concat out [leg])
  match List.fold step (Map.empty, []) legs:
    case (_, out): out

let groupByDest acc leg =
  Map.add leg.dest (List.concat (Option.withDefault [] (Map.tryFind leg.dest acc)) [leg]) acc

let insertByDep leg sorted =
  match sorted:
    case []: [leg]
    case [x, *rest]:
      if leg.depMin <= x.depMin then List.concat [leg] sorted
      else List.concat [x] (insertByDep leg rest)

let sortLegs legs = List.fold (fun acc l -> insertByDep l acc) [] legs

let minOf xs =
  match xs:
    case []: 0
    case [x]: x
    case [x, *rest]:
      let m = minOf rest
      if x < m then x else m

let maxOf xs =
  match xs:
    case []: 0
    case [x]: x
    case [x, *rest]:
      let m = maxOf rest
      if x > m then x else m

# --- title-casing (feed descriptions are UPPERCASE) ---
# Uppercase the first *alphabetic* char in a char list, leave the rest. The test
# `String.upper c != c` is true only for a lowercase letter, so "(south"
# capitalizes the s, not the "(".
let capFirst chars =
  match chars:
    case []: []
    case [c, *rest]:
      if String.upper c != c then List.concat [String.upper c] rest
      else List.concat [c] (capFirst rest)

let capWord w = String.join "" (capFirst (String.toList w))
let titleHyphen w = String.join "-" (List.map capWord (String.split "-" w))
let titleCase name = String.join " " (List.map titleHyphen (String.split " " (String.lower name)))

# --- rendering (Markdown) ---
let nameOf names tp = titleCase (Option.withDefault tp (Map.tryFind tp names))
let plural n word = if n == 1 then f"{n} {word}" else f"{n} {word}s"
let rangeStr lo hi = if lo == hi then f"{lo} min" else f"{lo}-{hi} min"

let renderService names leg =
  let stopStr =
    match leg.via:
      case []: "direct"
      case _:
        let viaStr = String.join ", " (List.map (nameOf names) leg.via)
        f"via {viaStr}"
  let onward = if leg.dest == leg.finalDest then "" else f" _(to {nameOf names leg.finalDest})_"
  f"- **{showTime leg.depMin} → {showTime leg.arrMin}** · {leg.jt} min · {stopStr}{onward}"

let renderGroup names g =
  match g:
    case (dest, legs):
      let times = List.map (fun l -> l.jt) legs
      let tr = plural (List.len legs) "train"
      let header = f"### {nameOf names dest} — {tr}, {rangeStr (minOf times) (maxOf times)}"
      let body = String.join "\n" (List.map (renderService names) (sortLegs legs))
      f"{header}\n\n{body}"

let daySection names label day allLegs =
  let legs = dedupLegs (List.filter (fun l -> runsOn day l.rdays) allLegs)
  let pairs = List.map (fun g -> match g: case (dest, ls): (nameOf names dest, renderGroup names g))
                       (Map.toList (List.fold groupByDest Map.empty legs))
  let body = String.join "\n\n" (List.map (fun p -> match p: case (nm, txt): txt) (List.sort pairs))
  f"## {label}\n\n{body}"

let buildReport names raw =
  let allLegs = List.fold (fun acc r -> List.concat acc (legsOfRun r)) [] (List.map toRun raw)
  let intro = "# Direct services from Chippenham\n\n_Every station reachable directly (no change) from Chippenham, grouped by destination, in departure order. Counts reflect the weekly day-of-week pattern of permanent + new schedules minus cancellations; a specific date may differ._"
  let wd = daySection names "Monday–Friday (representative day: Tuesday)" 1 allLegs
  let sa = daySection names "Saturday" 5 allLegs
  let su = daySection names "Sunday" 6 allLegs
  String.join "\n\n" [intro, wd, sa, su]

let _ =
  match scan dataFile:
    case (names, raw): writeText (toPath outFile) (buildReport names raw)

print (f"Wrote Markdown direct Chippenham services to {outFile}")

# The SAME timetable as chippenham.pyfun, but the streaming boundary is a native
# Python loop (nr_stream.py) instead of a Pyfun `Seq.fold`. The domain engine below
# is byte-for-byte identical; only the I/O at the top differs.
#
#   cargo run -- compile examples/interop/network-rail/chippenham_fast.pyfun \
#       -o examples/interop/network-rail/chippenham_fast.py
#   PYTHONPATH=examples/interop/network-rail \
#       python examples/interop/network-rail/chippenham_fast.py
#
# Why keep this alongside the pure-Pyfun version? Because it's ~15x faster on the
# real feed (~5s vs ~75s for ~20M lines), and that gap is the whole lesson: the hot
# per-line loop belongs in a tight, C-backed Python `for` loop, not in a Pyfun fold
# that pays a call per element. Pyfun's job is the typed engine; Python's optimized
# libraries do the grinding. `extern` is the boundary that lets you have both —
# leverage the fast thing that already exists rather than re-emulate it in Pyfun.

=== counter.pyfun ===
# A page that takes input: a counter with a button, running on Pyodide.
#
#   pyfun bundle examples/interop/browser/counter.pyfun -o site --page examples/interop/browser/counter.html
#   python -m http.server -d site     # then open http://localhost:8000/
#
# `Dom` is the typed façade in dom.pyfun; this program is what a consumer of it
# looks like. The state is a top-level `let mut` that the click handler
# reassigns (a `global` in the emitted Python); the handler is an ordinary Pyfun
# function whose one parameter is the event, handed to the page through
# `Dom.proxy`. No callback convention to remember: Python calls `bump(event)` with
# one argument, and that is the one parameter `bump` has.

import Dom

let mut count = 0

let render _ =
  Dom.setText (Dom.byId "count") f"{count}"

let bump _ =
  count <- count + 1
  render ()

Dom.on (Dom.byId "plus") "click" (Dom.proxy bump)
render ()
print "counter ready"
=== datetime.pyfun ===
# datetime — the most-imported stdlib module, wrapped as typed, *pure* values.
#
#   pyfun run examples/interop/datetime.pyfun
#
# datetime is all classes, which makes it a tour of the object-side extern forms:
# a class target as a constructor, property/method receivers, and Python's
# `operator` module for the arithmetic the classes define with `+`/`-`. Building
# and combining dates is deterministic, so these externs are declared `pure` —
# the boundary's opt-out — and `let pure` then *proves* a whole pipeline effect-free.

extern type Datetime
extern type Delta

# A class is callable, so a class target is a constructor: `date 2026 7 13` lowers
# to datetime.datetime(2026, 7, 13), auto-importing the class's module.
extern pure date : int -> int -> int -> Datetime = datetime.datetime
extern pure days : int -> Delta = datetime.timedelta      # timedelta(days) — first positional

# Classmethods sit behind the *lowercase* class, which the import heuristic would
# read as a submodule — an `extern import` declares the module explicitly, exactly
# like Python's own import statement (DESIGN.md §6). `now` stays a plain (io)
# extern: a fresh value each call is an effect, and `let pure` over it would be a
# compile error; parsing a fixed string is pure.
extern import datetime
extern now : unit -> Datetime = datetime.datetime.now
extern pure parseIso : string -> Datetime = datetime.datetime.fromisoformat

# Members of the opaque handles: `= .attr` reads a property, `= .attr()` calls a method.
extern pure weekday   : Datetime -> int    = .weekday()   # 0 = Monday
extern pure isoformat : Datetime -> string = .isoformat()
extern pure inDays    : Delta -> int       = .days

# datetime's arithmetic lives on `+`/`-`, and Python exposes every operator as a
# plain function in the `operator` module — ready-made extern targets.
extern pure shift   : Datetime -> Delta -> Datetime = operator.add   # d + delta
extern pure between : Datetime -> Datetime -> Delta = operator.sub   # b - a

let launch = date 2026 7 13
print (isoformat launch)                                # 2026-07-13T00:00:00
print (weekday launch)                                  # 0 — a Monday
print (isoformat (parseIso "2026-07-13"))               # 2026-07-13T00:00:00

# `now` performs io, so `sinceLaunch` is inferred io — and prints nothing fixed.
let sinceLaunch = inDays (between (now ()) launch)

# The externs are pure, so the compiler verifies this pipeline touches no effect.
let pure daysBetween a b = inDays (between b a)
print (daysBetween launch (shift launch (days 90)))     # 90

# Boundary handles flow through the ordinary combinators like any other value.
let sprintEnds = List.map (fun n -> isoformat (shift launch (days (n * 14)))) [1, 2, 3]
print sprintEnds

# An impossible date raises ValueError in Python; `try` reifies it as a Result,
# and the match must handle both outcomes.
let describe r =
  match r:
    case Ok d: f"valid: {isoformat d}"
    case Error e: f"invalid: {e.errorKind}"

print (describe (try (date 2026 2 30)))                 # invalid: ValueError
=== dom.pyfun ===
# A typed façade over the page, for a program that runs in the browser under
# Pyodide (`pyfun bundle`). Pyodide exposes the page as the `js` module, so the
# DOM is a set of externs like any other Python library; this module writes them
# once so a program imports them (`import Dom`) rather than repeating them.
#
# Every handle is opaque: the compiler follows an `Element` through the program
# without looking inside it. A callback handed to the page must be wrapped with
# `proxy` (Pyodide's `create_proxy`), which keeps it alive for as long as the page
# holds it; the façade is where that detail lives, not the program.

extern import js
extern import pyodide.ffi

extern type Element
extern type Event
extern type Callback

# The document, by id; a missing id raises at the boundary like any extern.
extern byId: string -> Element = js.document.getElementById

# Replace an element's children with one text node (`el.replaceChildren(text)`).
extern setText: Element -> string -> unit = .replaceChildren()

# `el.setAttribute(name, value)` and `el.append(child)`.
extern setAttr: Element -> string -> string -> unit = .setAttribute()
extern append: Element -> Element -> unit = .append()
extern create: string -> Element = js.document.createElement

# `el.addEventListener(event, callback)`; the callback performs `io` (it will).
extern on: Element -> string -> Callback -> unit = .addEventListener()
extern proxy: (Event ->{io} unit) -> Callback = pyodide.ffi.create_proxy
=== hello.pyfun ===
# A taste of the type-checked Phase 3 subset that compiles to runnable Python.
#   pyfun check   examples/hello.pyfun     (type-check)
#   pyfun compile examples/hello.pyfun     (emit Python)
#   pyfun run     examples/hello.pyfun     (run it)

# Algebraic data types: a plain enum. (`Option a = None | Some a` is built in.)
type Color = Red | Green | Blue

# Records: named-field product types (nominal; field names are unique).
type Point = { x: int, y: int }

# Curried function: inferred type int -> int -> int. `pure` is an opt-in
# assertion that the function performs no effects (no `io`); the compiler infers
# effects everywhere and only checks them where you ask. Effects are erased at
# lowering. (`let pure shout n = print n` would be a compile error.)
let pure add a b = a + b

# Partial application (functools.partial); full application (add(1, 4)).
let inc = add 1
let five = 4 |> inc

# Recursion is implicit: a function is in scope in its own body, like Python's
# `def` (no `rec` keyword). Deep recursion shares Python's stack limit; use the
# `List`/`Seq` combinators for stack-safe bulk work.
let fact n =
  if n == 0 then 1
  else n * fact (n - 1)

# Standard combinators are built in (`DESIGN.md` §6): `id : 'a -> 'a`,
# `const : 'a -> 'b -> 'a`, `ignore : 'a -> unit`, `flip` swaps a function's
# first two arguments. `flip` is effect-polymorphic; the rest are pure. (A user
# `let id x = x` would shadow the builtin, like any other prelude name.)
let same = id 42                       # 42
let always7 = const 7 "discarded"      # 7
let subtract a b = a - b
let backwards = flip subtract 3 10     # subtract 10 3 = 7
let zeroed = List.map (const 0) [1, 2, 3]   # [0, 0, 0]
let dropped = ignore 42                # unit (None) — ignore discards its argument

# Doc comments: a `##` line at column 0 attaches to the following top-level
# declaration as documentation (LSP hover shows it below the type), unlike an
# ordinary `#` comment. It erases at compile time.
## Returns 1 when the flag is set, else 0.
let pick b = if b then 1 else 0

# Multi-line doc comment: consecutive `##` lines join into one doc block (with
# `\n` between them), attached to the declaration that follows — here a `type`.
# This sum also shows the offside layout — one `| Variant` per line in an indented
# block after `=` (the F#/ML habit); `Color` above uses the inline `A | B | C`
# form. Both parse identically (a leading `|` is optional; the printer
# canonicalizes back to one line).
## A traffic signal.
## `Walk` when clear, `Wait` while it counts down, `Stop` otherwise. Ordered by
## declaration, so `Walk < Wait < Stop` under the derived comparison.
type Signal =
  | Walk
  | Wait
  | Stop

# Typed holes (shown as a comment — a hole blocks compilation by design, so this
# file can still `run`): write `?` or `?name` in expression position and the
# compiler reports the type it expects there — plus in-scope bindings that fit
# directly (`try:`) and functions whose result fits, applied to more holes
# (`or:`) — for type-driven development. E.g.
#     let use = String.concat ?g "!"
# makes `pyfun check` report `note: hole ?g has type string — try: greeting — or:
# String.upper ?, String.fromInt ?` and hover show it; `compile`/`run` refuse
# until it's filled. Fill inward from the types (and the suggested fits).

# Blocks: an indented body is a sequence of statements ending in the value. They
# open after `=` (a `let` body), `->` (a lambda), `case P:` (a `match` arm), `then`/`else`.
# `let mut` + `<-` give local mutation; reassigning a plain `let` is a compile
# error (immutable by default). Lowers to a flat Python statement sequence.
let scaled a b =
  let mut acc = a
  acc <- acc + b
  acc <- acc * 2
  acc

# A closure that reassigns a captured `mut` — `tally` mutates `running` across
# calls. The compiler emits Python's `nonlocal`/`global` so the capture is correct
# (F# auto-boxes a captured mutable into a reference cell; same idea).
let accumulate =
  let mut running = 0
  let tally x =
    running <- running + x
    running
  let first = tally 10
  tally 5                             # 15  (10 then +5)

# A block in a `match` arm (opened after `case P:`).
let describeSign n =
  match n:
    case 0: "zero"
    case _:
        let positive = n > 0
        if positive then "positive" else "negative"

# Exhaustive pattern matching over an ADT (checked at compile time).
let unwrap o =
  match o:
    case Some v: v
    case None: 0

let answer = unwrap (Some 42)

# An as-pattern (`p as x`) binds the whole matched value alongside destructuring —
# and it's transparent to the exhaustiveness check (`Some v as w` covers Some).
let doubleIfSome o =
  match o:
    case Some v as w: w
    case None: None
let stillFortyTwo = doubleIfSome (Some 42)   # Some 42

let rank c =
  match c:
    case Red: 1
    case Green: 2
    case Blue: 3

# Mutual recursion works implicitly (no `rec`/`and`): `isEven` sees `isOdd` and
# vice versa, defined in either order.
let isEven n = if n == 0 then true else isOdd (n - 1)
let isOdd n = if n == 0 then false else isEven (n - 1)
let tenEven = isEven 10               # true

# Or-patterns: `case a | b:` matches any alternative (Python-identical). All
# alternatives must bind the same variables (here, none).
let size n =
  match n:
    case 1 | 2 | 3: "small"
    case _: "big"

# Guards: a `case` arm may carry an `if` condition. A guarded arm is refutable —
# it never counts toward exhaustiveness, so a catch-all `case _` is still needed.
let sign n =
  match n:
    case 0: "zero"
    case m if m > 0: "positive"
    case _: "negative"

# `elif` — sugar for `else if`; `if`/`then`/`else` stay (an inline expression, not
# `:`-framed). The printer canonicalizes any else-if chain to `elif`.
let grade n =
  if n >= 90 then "A"
  elif n >= 80 then "B"
  elif n >= 70 then "C"
  else "F"

# Higher-order composition: ('b -> 'c) -> ('a -> 'b) -> 'a -> 'c.
let compose = fun f g x -> f (g x)
let inc2 = compose inc inc
let seven = inc2 five

# Function composition operators: `>>` is left-to-right (f then g), `<<` is
# right-to-left (math ∘). Tighter than `|>`, so `x |> f >> g` = `x |> (f >> g)`.
let twice x = x * 2
let incThenDouble = inc >> twice
let doubleThenInc = inc << twice
let eight = incThenDouble 3            # twice (inc 3) = 8
let sevenC = doubleThenInc 3           # inc (twice 3) = 7
# Backward pipe `<|` applies a function to an argument (F#'s `<|`), dropping parens
# on the right: `inc <| twice 3` = `inc (twice 3)`. Right-associative.
let sevenB = inc <| twice 3            # inc (twice 3) = 7

# In-file modules: a namespace of `let` bindings. Members see each other
# unqualified inside (`area` calls `square`); outside they are `Geom.member`.
module Geom =
  let square x = x * x
  let area w h = w * h
  let diagSq w h = square w + square h
let floorArea = Geom.area 4 5         # 20
let diag = Geom.diagSq 3 4            # 25

# Computation expressions (F# showcase features):
#   result {} — railway-oriented, short-circuits on Error (built-in Ok/Error).
let checked ok v =
  result {
    let! x = if ok then Ok v else Error 0
    return x + 1
  }

#   option {} — the same railway over Option, for chaining lookups that may find
#   nothing. One `case None` arm covers the whole chain instead of one per step.
let firstTwo xs =
  option {
    let! a = List.get 0 xs
    let! b = List.get 1 xs
    return a + b
  }
print (firstTwo [10, 20, 30])      # Some(30)
print (firstTwo [10])              # None_ — the emitted class name, `None` being taken in Python

#   seq {} — a lazy sequence, lowered to a Python generator.
let small =
  seq {
    yield 1
    yield 2
    yield 3
  }

#   async {} — lowered to a Python coroutine (async def / await). A coroutine must be
#   awaited somewhere, so run one to a value with an `extern` over `asyncio.run`.
extern runAsync: Async a -> a = asyncio.run
let computed =
  async {
    let! x = async { return 20 }
    return x + 1
  }
print (runAsync computed)          # 21

# User-defined CE builders: any in-file `module` providing `bind`/`return_` (and
# `yield_`/`combine`/`delay`/… for sequence builders) is a builder. `Maybe { … }`
# desugars to `Maybe.bind`/`Maybe.return_` calls — ordinary inference does the rest.
module Maybe =
  let bind m f =
    match m:
      case Some x: f x
      case None: None
  let return_ x = Some x
let addOpt a b =
  Maybe {
    let! x = a
    let! y = b
    return x + y
  }

# Tuples: structural (anonymous) products, `(a, b, c)`. `()` is unit and `(x)` is
# grouping, so a tuple has two or more elements. They lower ~1:1 to Python tuples
# and destructure in `match` (a single variable pattern is exhaustive).
let pairUp a b = (a, b)
let origin = (0, 0)
let swap p =
  match p:
    case (a, b): (b, a)
let labelled = (1, "one", true)              # (int, string, bool)
let firstOf t =
  match t:
    case (a, _, _): a

# Units of measure: dimensional analysis at compile time, erased at runtime.
measure m
measure s
measure kg

# Derived-measure aliases name a compound of base units (and may build on each
# other). They expand to base units, so `<N>` and `<kg m / s^2>` are the same type.
measure N = kg m / s^2               # newton
measure Pa = N / m^2                 # pascal, built on N

let push = 10<N>
let panel = 2<m^2>
let stress = push / panel            # float<Pa>, dimensionally N/m^2

let distance = 100<m>
let elapsed = 10<s>
let speed = distance / elapsed       # true division -> float<m/s>
let trip = distance // 4<m>          # floor division -> int (dimensionless)
let leftover = distance % 30<m>      # modulo -> int<m> (unit-preserving); 100 % 30 = 10

# `area` is unit-polymorphic: int<'u> -> int<'v> -> int<'u 'v>.
let area w h = w * h
let floor = area 4<m> 5<m>            # int<m^2>
# `distance + elapsed` would be a compile error: int<m> vs int<s>.

# Unit-aware roots: sqrt : float<'u^2> -> float<'u> (√area = length) and
# cbrt : float<'u^3> -> float<'u> (∛volume = length). The unit's exponents halve
# (sqrt) or divide by 3 (cbrt); a non-square/non-cube unit is a compile error
# (`sqrt 2.0<m>`, `cbrt 2.0<m^2>` are rejected dimensionally). These two are the
# only tractable unit-carrying power ops — `**` stays dimensionless.
let side = sqrt 16.0<m^2>            # float<m> — 4.0, unit halved (m^2 -> m)
let edge = cbrt 64.0<m^3>            # float<m> — 4.0, unit thirded (m^3 -> m)

# Float literals support scientific notation (`1e6`, `2.5e-3`) — the exponent sign
# is part of the literal. Physics constants read naturally, units and all:
let lightSpeed = 3.0e8<m/s>          # float<m/s>
let planck = 6.626e-34               # float

# Prefix negation `-e`: numeric and unit-preserving. It's a prefix operator (not a
# negative-literal token), so `-` is subtraction with a left operand and negation
# without — `depth - 2` subtracts, `-depth` negates.
let depth = -20<m>                    # int<m>, a negative literal
let rebound = -(3 + 4)               # -7 (a looser operand is parenthesized)

# Numeric conversions round/floor/ceil/truncate : float<'u> -> int<'u>, unit-
# preserving like abs/min/max (round is a Python builtin; the rest use `math`).
let rounded = round 3.7              # 4
# (`floor` is shadowed below by a user binding of the same name — user
# definitions shadow prelude names — so this uses `truncate`.)
let truncated = truncate speed       # 10   (float<m/s> -> int<m/s>, unit kept)
# String.toFloat is a total parse (Option float), mirroring String.toInt.
let parsedF = Option.withDefault 0.0 (String.toFloat "2.71828")  # 2.71828

# Exponentiation `**` is float-only, right-associative, and binds tighter than
# unary minus (so `-2.0 ** 2.0` is `-(2.0 ** 2.0)`, like Python).
let cube = 3.0 ** 3.0                # 27.0
let tower = 2.0 ** 3.0 ** 2.0        # 512.0  (right-assoc: 2 ^ (3 ^ 2))
let negPow = -2.0 ** 2.0            # -4.0   (`**` tighter than unary minus)

# Numeric-literal ergonomics: `_` digit separators and 0x/0o/0b bases.
let million = 1_000_000             # 1000000
let mask = 0xFF                     # 255  (hex; also 0o17 octal, 0b1010 binary)

# Externs: typed imports of arbitrary Python callables (`DESIGN.md` §6). The
# Pyfun name equals the Python name unless a `= target` clause gives a dotted
# path (which is imported automatically). Type variables are bare lowercase, as
# in `type` decls. The boundary is effectful-by-default — a plain `extern` is
# `io` at full application — unless asserted `pure`; `math.fabs` is pure.
# (`extern sqrt`/`extern cbrt` would clash — sqrt/cbrt are the built-in unit-aware
# roots now, so a Python name that isn't a Pyfun builtin is used here.)
extern show: a -> string = str
extern pure fabs: float -> float = math.fabs

# Declared-arrow effect annotations: effects are inferred everywhere, but an
# `extern` (or declared `type`) arrow may state effect labels explicitly with
# `->{label, …}` — beyond the default inferred `io`, e.g. the `async` label.
# An async Python client returns a coroutine, so its result is the `Async`
# type and `let!` inside an `async { }` block is where it is awaited; an
# `->{async}` arrow that returns anything else is rejected. (Uncalled here, so
# it erases; calling `sleepAsync` would make its caller perform `async`, so it
# couldn't sit inside a `let pure`.)
extern sleepAsync: float ->{async} Async unit = asyncio.sleep

let label = show 42                  # "42"  (str(42))
let root = fabs (0.0 - 3.0)          # 3.0   (math.fabs(-3.0))

# `try e` catches a Python exception (e.g. from an `extern`) into a `Result a
# Exception`, so the imperative FFI boundary becomes the FP error type. The
# reserved `Exception` record exposes `errorKind` (the class name) and
# `errorMessage`. `try` binds looser than `+` but tighter than `|>`.
extern parseInt: string -> int = int
let parsed42 = Result.withDefault 0 (try (parseInt "42"))    # 42
let parsedBad = Result.withDefault 0 (try (parseInt "oops")) # 0  (ValueError caught)
let errorKindOf s =
  match try (parseInt s):
    case Ok n: "ok"
    case Error e: e.errorKind                                # string literal patterns

# Lists: the eager `List a` collection (lowers to a Python list). Literals are
# comma-separated `[a, b, c]`. Collection functions are module-qualified
# (`List.map`, `Set.add`, `Map.tryFind`) — the module disambiguates a shared name
# like `len`, so no overloading is needed. The higher-order list ops are
# effect-polymorphic (mapping an impure function makes the call impure).
let ns = [1, 2, 3, 4, 5]
let squares = List.map (fun x -> x * x) ns   # [1, 4, 9, 16, 25]
let bigs = List.filter (fun x -> x > 2) ns   # [3, 4, 5]
let total = List.fold (fun a b -> a + b) 0 ns # 15
let count = List.len ns                      # 5

# An operator section `(op)` is a binary operator as a curried function, so the
# fold above shortens to `(+)`; `(*) 2` partially applies it (F#-style).
let product = List.fold (*) 1 ns             # 120
let doubled = List.map ((*) 2) ns            # [2, 4, 6, 8, 10]

# Completeness ops, each with big-O honest to a Python list (an array): `get` is
# O(1) and total (returns `Option`, so no IndexError — no `xs[0]` syntax exists);
# `contains` is O(n) (use `Set` for O(1)); `concat` O(n+m); `sort` O(n log n);
# `find` O(n) first-match. There's no cheap prepend — that's the linked-list non-goal.
let third = Option.withDefault 0 (List.get 2 ns)   # 3 (0-based)
let missing = Option.withDefault 0 (List.get 9 ns) # 0 (out of bounds -> None)
let hasThree = List.contains 3 ns            # true
let joined = List.concat ns [6, 7]           # [1, 2, 3, 4, 5, 6, 7]
let ordered = List.sort [3, 1, 2]            # [1, 2, 3]
# Derived ordering: user sum types / records / tuples compare structurally, so
# `<` (and `List.sort`) work on them. A sum orders by variant declaration order
# then field-by-field (Red < Green < Blue).
let sortedColors = List.sort [Blue, Red, Green]  # [Red, Green, Blue]
let redFirst = Red < Green                       # true
# Built-in Option/Result order too: None < Some, Ok < Error (payloads orderable).
let sortedOpts = List.sort [Some 3, None, Some 1]  # [None, Some 1, Some 3]
let firstBig = List.find (fun x -> x > 3) ns # Some 4

# Sets and maps lower to Python `set` / `dict`; `Map.tryFind` returns the built-in
# `Option`, consumed with the `Option` module.
let uniq = Set.ofList [1, 1, 2, 3]           # {1, 2, 3}
let scores = Map.add "ada" 10 (Map.add "alan" 9 Map.empty)
let ada = Option.withDefault 0 (Map.tryFind "ada" scores)   # 10
let nobody = Option.withDefault 0 (Map.tryFind "x" scores)  # 0

# The `Option` module chains lookups: `bind` sequences Option-returning steps,
# `filter` keeps a value only if it passes, `toResult` bridges to `Result`.
let doubled = Option.bind (fun n -> Some (n * 2)) (Map.tryFind "ada" scores)  # Some 20
let bigScore = Option.filter (fun n -> n > 9) (Map.tryFind "alan" scores)     # None (9 <= 9)
let adaOrErr = Option.toResult "no ada" (Map.tryFind "ada" scores)           # Ok 10

# `List.zip` pairs two lists into a list of tuples; `Map.ofList` builds a map from
# that pair list (and `Map.toList` is its inverse) — both ride on tuples.
let names = ["ada", "alan"]
let ages = [36, 41]
let byName = Map.ofList (List.zip names ages)
let adaAge = Option.withDefault 0 (Map.tryFind "ada" byName)  # 36
let entries = Map.toList byName               # [("ada", 36), ("alan", 41)]

# Strings: the `String` module over the built-in `string` type. All pure; there is
# no `char` type, so `String.toList` yields single-character strings. `String.toInt`
# is a total parse — it returns the built-in `Option`, `None` on failure.
let title = String.concat "Hello, " "World"   # "Hello, World"
let loud = String.upper title                 # "HELLO, WORLD"
let words = String.split ", " title           # ["Hello", "World"]
let csv = String.join "," words               # "Hello,World"
# `slice` is total Python slicing (clamps out of range); `tryIndexOf` returns an
# Option (None when absent), so neither can raise.
let sub = String.slice 0 5 title              # "Hello"
let commaAt = Option.withDefault (0 - 1) (String.tryIndexOf "," title)  # 5
let parsed = Option.withDefault 0 (String.toInt "41")   # 41
let unparsed = Option.withDefault 0 (String.toInt "x")  # 0
# Non-ASCII string literals are real UTF-8 (not one Latin-1 byte per char), so
# `String.len "café"` is 4, not 5 — and the emitted Python is correctly encoded.
let cafeLen = String.len "café"                # 4 (é is one character)
# Escapes: \" \\ \n \t \r and Rust-style \u{HEX} (decoded to the character).
let emojiLen = String.len "hi \u{1F600}"       # 4 ("hi " + 😀)
# Raw strings `r"..."` — no escape processing, backslashes are literal (handy for
# Windows paths and regex). `r"a\nb"` is 4 chars (a, \, n, b), not a newline.
let winPath = r"C:\Users\pyfun"                # C:\Users\pyfun (literal backslashes)
let rawLen = String.len r"a\nb"                # 4 (backslash and n are literal)

# String interpolation: `f"...{expr}..."` (Python-style). Holes take any full
# expression and are stringified; `{{`/`}}` are literal braces. Lowers 1:1 to a
# Python f-string (targets Python 3.12+).
let who = "Ada"
let scoreLine = f"{who} scored {parsed} ({String.upper who})"   # "Ada scored 41 (ADA)"
let sumLine = f"{parsed} doubled is {parsed + parsed}"          # "41 doubled is 82"
# Self-documenting `{expr=}` holes echo the source before the value (Python's rule):
let debugLine = f"{parsed=}"                                    # "parsed=41"
# Triple-quoted strings span lines (newlines and lone quotes are literal content;
# escapes still process): plain `"""..."""`, interpolated `f"""..."""` (holes as
# usual), and raw `r"""..."""` (backslashes literal too).
let banner = """== pyfun ==
a "quoted" word"""
let summary = f"""who: {who}
score: {parsed}"""
# Raw triple-quoted `r"""..."""`: spans lines AND leaves backslashes literal.
let rawBlock = r"""C:\logs
a\tb"""                                # backslashes + newline all literal

print leftover                       # 10    (modulo, unit-preserving)

print truncated                      # 10    (truncate, unit-preserving)
print side                           # 4.0   (unit-aware sqrt: float<m^2> -> float<m>)

print cube                            # 27.0  (exponentiation)

# Comparison (`comparison 'a => …`) and equality (`'a -> 'a -> bool`, structural
# for ADTs) both yield bool. `<` is comparison here because of the space; `5<m>`
# without a space is a unit annotation.
print (answer == 42)                 # True
print (seven < 10)                   # True
print (speed < 50<m/s>)              # True   (dimensional comparison)
print (Some 42 == Some answer)       # True   (structural ADT equality)
# Chained comparison (Python-style): `lo <= x < hi` means `lo <= x and x < hi`,
# evaluating `x` once. Lowers to Python's own chained comparison.
print (0 <= seven < 10)              # True   (chained comparison)

# Logical operators: `and` / `or` / `not` (Python keywords, Python precedence).
print (seven < 10 and answer == 42)  # True
print (not (seven == 8) or false)    # True

# Records: construct with `{ f = v }`, access with `.f`, copy-and-update with
# `{ r with f = v }`. The record type is resolved from the field names.
let here = Point { x = 3, y = 4 }
let shifted = { here with x = 10 }
let dot r = r.x + r.y                 # field access fixes `r : Point`
print here.x                          # 3
print shifted.x                       # 10
print (dot here)                      # 7
print here                            # Point(3, 4)

# Nested-update sugar: a dotted path `{ r with a.b = v }` updates a field deep in
# a record — sugar for `{ r with a = { r.a with b = v } }`. Arbitrary depth, and
# the base is evaluated once with sibling fields (here `start.y`, `finish`) copied.
type Segment = { start: Point, finish: Point }
let seg = Segment { start = Point { x = 0, y = 0 }, finish = here }
let moved = { seg with start.x = 99 }
print moved.start.x                   # 99  (updated through the path)
print moved.start.y                   # 0   (sibling preserved)
print moved.finish.x                  # 3   (untouched field preserved)
print (scaled 3 4)                    # 14   ((3 + 4) * 2, via local mutation)
print accumulate                      # 15   (closure capturing a mutable)

# Record patterns: destructure in `match`. `{ x }` shorthand binds the field; a
# pattern may name a subset of fields, and `{ x, y }` is an exhaustive catch-all.
let describe p =
  match p:
    case Point { x = 0, y = 0 }: "origin"
    case Point { x = 0 }: "y-axis"
    case Point { x, y }: "elsewhere"
print (describe (Point { x = 0, y = 0 }))   # origin
print (describe here)                 # elsewhere

# List/sequence patterns in `match` over a `List`: `[]`, a fixed `[x]`, and a
# `*rest` star that may sit ANYWHERE (Python's one-star rule) — `[x, *rest]`
# (head + tail), `[*init, last]` (all but the last), `[a, *mid, z]` (first and
# last, middle bound). `[]` plus a star arm is exhaustive.
let shape xs =
  match xs:
    case []: "empty"
    case [only]: "one"
    case [a, *mid, z]: "first and last, some middle"
let lastOf xs =
  match xs:
    case []: 0
    case [*init, last]: last
print (shape [1, 2, 3, 4])            # first and last, some middle
print (shape [7])                     # one
print (lastOf [10, 20, 30])           # 30  (non-last star: [*init, last])

# Active patterns (DESIGN.md 7.2.1): named recognizers whose cases are matched
# like constructors. A total pattern (|A|B|) is a closed case set, so its match
# needs no `_`; a partial pattern (|A|_|) returns Option (the case binds the
# payload) or bool (a pure predicate, binds nothing) and needs a fallthrough.
let (|Even|Odd|) n = if n % 2 == 0 then Even else Odd
let parity n =
  match n:
    case Even: "even"
    case Odd: "odd"

let (|Positive|_|) n = if n > 0 then Some n else None
let clamped n =
  match n:
    case Positive p: p
    case _: 0

let (|Blank|_|) s = String.strip s == ""
let lineKind s =
  match s:
    case Blank: "empty"
    case _: "text"

print (parity 4)                      # even

print (clamped 9)                     # 9
print (clamped (0 - 5))               # 0

# json — an untyped, throwing boundary becomes typed, total data.
#
#   pyfun run examples/interop/json_decode.pyfun

# "Parse, don't validate": give `json.loads` a Pyfun type at the call site, and `try`
# turns its exceptions into a `Result` you must handle — a parse failure is a value,
# never a crash. (For decoding into your own records, see json_to_adt.pyfun.)

# A JSON array of one type decodes to a `List`; a flat object to a `Map`. `= json.loads`
# auto-imports `json`; the boundary is effectful, so both are `io`.

    case Error e: 0.0

=== json_to_adt.pyfun ===
# Decode (built-in) — JSON into your own typed records, totally. THE showcase.
#
#   pyfun run examples/interop/json_to_adt.pyfun
#
# The lesson every language teaches on JSON is "parse, don't validate": turn untrusted,
# dynamically-typed input into your *own* typed data at the edge, so the rest of the
# program can't be fed a wrong shape. Pyfun ships an Elm-style `Decode` module — a
# `Decoder a` is a pure, total recipe, and `Decode.decodeString` runs it over a JSON
# string to yield `Result a Exception`. Bad input is a value you handle, never a crash.

type User = { name: string, age: int, roles: List string }

# A decoder is built compositionally: `Decode.field` pulls one object field and runs
# a decoder on it; `Decode.map3` combines three field decoders into one that builds
# the record. No hand-written field access, no per-field error handling — the module
# threads the `Result` railway for you. `Decode.string`/`int` are strict: a JSON
# number where a string is expected is an `Error`, not a silent coercion.
let userDecoder =
  Decode.map3 (fun name age roles -> User { name = name, age = age, roles = roles })
    (Decode.field "name" Decode.string)
    (Decode.field "age" Decode.int)
    (Decode.field "roles" (Decode.list Decode.string))

# Consuming the result is an exhaustive `match` — the compiler makes you handle both
# outcomes. `Exception` exposes `errorKind` (the Python class name) and `errorMessage`.
let describe r =
  match r:
    case Ok u: f"""{u.name} ({u.age}): {String.join ", " u.roles}"""
    case Error e: f"decode failed ({e.errorKind}): {e.errorMessage}"

# A well-formed object decodes to a typed `User`, array field and all.
print (describe (Decode.decodeString userDecoder """{"name": "ada", "age": 36, "roles": ["admin", "dev"]}"""))
# A missing field short-circuits — no crash, a value describing the failure.
print (describe (Decode.decodeString userDecoder """{"name": "bob", "age": 40}"""))
# A wrong type is caught the same way (age is a string, not a number).
print (describe (Decode.decodeString userDecoder """{"name": "cy", "age": "old", "roles": []}"""))
# So is malformed JSON.
print (describe (Decode.decodeString userDecoder """not json at all"""))

# Output:
#   ada (36): admin, dev
#   decode failed (KeyError): 'roles'
#   decode failed (ValueError): expected an int, got str
#   decode failed (JSONDecodeError): Expecting value: line 1 column 1 (char 0)
#
# The whole pipeline is PURE (no `io`) — a `let pure` over `Decode.decodeString` type-
# checks, because the curated module knows JSON parsing has no side effects (the raw
# `extern json.loads` boundary in json_decode.pyfun is conservatively `io`). More
# combinators: `Decode.map`/`map2`/`map4`, `Decode.nullable` (JSON null → `Option`),
# `Decode.oneOf` (first decoder that succeeds — decode a union), `Decode.andThen`
# (a decoder chosen from an already-decoded value), `Decode.succeed`/`Decode.fail`.

# pathlib — file I/O, with effects inferred and failure made total.
#
#   pyfun run examples/interop/read_files.pyfun
#
# The compiler infers these calls are `io` and propagates it outward with no
# annotation, and `try` turns a missing-file error into a `Result`. Runs offline
# and cleans up after itself.

extern type Path
extern toPath: string -> Path = pathlib.Path
extern writeText: Path -> string -> int = pathlib.Path.write_text
extern readText: Path -> string = pathlib.Path.read_text
extern unlink: Path -> unit = pathlib.Path.unlink

# `roundTrip` calls `io` externs, so it is inferred `io` too — a `let pure roundTrip`
# would be a compile error. Purity is checked, not hoped for.
let roundTrip name =
  let p = toPath name
  let _ = writeText p "hello from pyfun"
  let back = readText p
  let _ = unlink p
  back

# Reading a missing file is total: `try` catches FileNotFoundError into `Error`.
let readOrNote name =
  match try (readText (toPath name)):
    case Ok s: s
    case Error e: f"(could not read {name}: {e.errorKind})"

print (roundTrip "pyfun_interop_demo.txt")   # hello from pyfun
print (readOrNote "no_such_file.txt")        # (could not read no_such_file.txt: FileNotFoundError)

# sqlite3 — a stateful, method-based (OOP) library, called safely from Pyfun.
#
#   pyfun run examples/interop/sqlite_query.pyfun
#
# Two patterns bridge an object-oriented library: an `extern type` for each opaque
# handle (a Connection/Cursor Pyfun never looks inside), and instance-method externs
# (`= .method()`) that call a method on their first argument — the receiver.

extern type Conn
extern type Cursor

extern connect: string -> Conn = sqlite3.connect        # a module function
extern execute: Conn -> string -> Cursor = .execute()   # conn.execute(sql)
extern fetchall: Cursor -> List a = .fetchall()         # cursor.fetchall()

# `let _ = e` runs a statement for its effect and discards the cursor it returns.
let db = connect ":memory:"
let _ = execute db "create table t (x int)"
let _ = execute db "insert into t values (1), (2), (3)"

# Rows come back as Pyfun tuples; a two-column row is a 2-tuple you destructure.
let firstCol r =
  match r:
    case (x, _): x

let rows = fetchall (execute db "select x, x * x from t order by x")
let total = List.fold (fun acc r -> acc + firstCol r) 0 rows

print rows                                # [(1, 1), (2, 4), (3, 9)]
print total                               # 6
=== structured_concurrency.pyfun ===
# asyncio — structured concurrency with `Task.scope`/`Task.start`, and a mailbox
# agent over `asyncio.Queue`, the F# `MailboxProcessor` shape.
#
#   pyfun run examples/interop/structured_concurrency.pyfun
#
# Runs offline. Every task is owned by a scope: `Task.scope` opens an
# `asyncio.TaskGroup` around an async body and does not return until every task
# started with `Task.start` has finished (or cancels them all if one fails).
# `Task.start` performs the `spawn` effect, and a `Task.scope` body is the only
# place that effect is discharged, so there is no way to start a task nothing
# will join, and no `cancel` to forget.

extern import asyncio
extern runAsync: Async a -> a = asyncio.run

# The mailbox: an asyncio queue as an opaque handle. `put_nowait` posts from
# anywhere; `get` is a coroutine, so its result is the `Async` type and `let!`
# is where the agent waits for the next message.
extern type Queue
extern newQueue: unit -> Queue = asyncio.Queue
extern post: Queue -> a -> unit = .put_nowait
extern take: Queue -> Async a = .get()

# The protocol is one ADT: spectators, reconnection or a clock are new
# constructors, not new plumbing.
type Msg = Said string | Tick | Quit

# One step of the agent's state: an effect and the next count. Sync, so the
# recursive call below stays a *direct* tail call.
let heard count s =
  print f"agent heard {s}"
  count + 1

let finished count = async { return count }

# The agent: one loop that is a `match` over the message, the loop's state
# threaded through the recursion. The self tail call compiles to a `while True:`
# inside the one coroutine (`DESIGN.md` §5.4), so an agent that handles a
# message a second for a year is still on its first stack frame.
let agentLoop inbox count =
  async {
    let! msg = take inbox
    return! (match msg:
      case Said s: agentLoop inbox (heard count s)
      case Tick: agentLoop inbox count
      case Quit: finished count)
  }

# A producer posts from its own task. The block ends in a plain call, which is
# `return ()` for a fire-and-forget block.
let producer inbox =
  async {
    for s in ["hello", "world"]:
      do! Async.sleep 0.01
      post inbox (Said s)
    post inbox Tick
    post inbox Quit
  }

# The session: the producer runs as a task of the scope while the agent awaits
# in the body; the closing of the scope joins the producer. Nothing here can
# leak past the end of `session`.
let session =
  Task.scope (async {
    let inbox = newQueue ()
    Task.start (producer inbox)
    let! count = agentLoop inbox 0
    return count
  })

let count = runAsync session
print f"agent handled {count} messages"

# `Task.start` performs `spawn`, and only a `Task.scope` body discharges it, so a
# start outside a scope is a compile error ("this starts a task outside any
# `Task.scope`"), not a leaked task at run time.

# Typed declarations for Python's `textwrap`, imported in Pyfun as `Wrap`.
# Every function here is pure: it reads its arguments and returns a string.
#
# A façade is ordinary Pyfun, so beside the `extern`s it can hold small

# last, so `text |> Wrap.fill 24` reads left to right.

extern import textwrap

extern pure shortenText: string -> int -> string = textwrap.shorten(width = ...)
extern pure indentText: string -> string -> string = textwrap.indent

# `text |> Wrap.fill 24`: the text re-flowed to lines of at most 24 characters.
let fill width text = fillText text width

# `text |> Wrap.shorten 30`: the text collapsed and cut to fit 30 characters,
# ending in " [...]" when something was cut.
let shorten width text = shortenText text width

# `text |> Wrap.indent "> "`: every non-blank line prefixed.
let indent prefix text = indentText text prefix
